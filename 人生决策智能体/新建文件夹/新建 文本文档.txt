## 1. API 基础路径封装

避免不同文件里手写：

```ts
fetch('/kline-tide/package')
fetch('/consult/records/:id/behavior')
```

应统一走 `apiClient`。

```ts
// app/src/api/httpClient.ts

const API_BASE =
  import.meta.env.VITE_API_BASE_URL ||
  (import.meta.env.PROD ? '/life/api/v1' : 'http://localhost:30001/api/v1');

export async function apiGet<T>(path: string, query?: Record<string, unknown>): Promise<T> {
  const url = new URL(`${API_BASE}${path}`, window.location.origin);

  Object.entries(query ?? {}).forEach(([key, value]) => {
    if (value !== undefined && value !== null && value !== '') {
      url.searchParams.set(key, String(value));
    }
  });

  const response = await fetch(url.toString());

  if (!response.ok) {
    throw new Error(`GET ${path} failed: ${response.status}`);
  }

  return response.json();
}

export async function apiPost<T>(
  path: string,
  body: Record<string, unknown>
): Promise<T> {
  const response = await fetch(`${API_BASE}${path}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body),
  });

  if (!response.ok) {
    throw new Error(`POST ${path} failed: ${response.status}`);
  }

  return response.json();
}
```

然后：

```ts
// app/src/services/klineTideApi.ts

import { apiGet } from '../api/httpClient';
import type { TidePackage } from '../types/tideClosedLoop';

export const klineTideApi = {
  fetchPackage(params = {}) {
    return apiGet<TidePackage>('/kline-tide/package', params);
  },
};
```

## 3. 月份标准化函数

现在最大隐患是：

```text
monthLabel = 2026年6月
date = 2026-06
query.month = 2026-6
```

这些格式不统一会导致 K线、雷达、相位无法对齐。

```ts
// app/src/utils/monthKey.ts

export function toMonthKey(input: string | number, month?: number): string {
  if (typeof input === 'number' && typeof month === 'number') {
    return `${input}-${String(month).padStart(2, '0')}`;
  }

  const str = String(input).trim();

  const matched = str.match(/(\d{4})\D?(\d{1,2})/);
  if (!matched) return str;

  const year = matched[1];
  const m = matched[2].padStart(2, '0');

  return `${year}-${m}`;
}
```

所有比较统一：

```ts
toMonthKey(kline.monthLabel) === toMonthKey(snapshot.date)
```

## 4. 图表点击事件防重复

Recharts 点击事件容易重复触发。
必须补一个简单去重。

```ts
// app/src/hooks/useEventDedup.ts

import { useRef } from 'react';

export function useEventDedup(delay = 500) {
  const lastKeyRef = useRef('');
  const lastTimeRef = useRef(0);

  return function shouldSkip(key: string) {
    const now = Date.now();

    if (lastKeyRef.current === key && now - lastTimeRef.current < delay) {
      return true;
    }

    lastKeyRef.current = key;
    lastTimeRef.current = now;
    return false;
  };
}
```

使用：

```ts
const shouldSkip = useEventDedup();

function handleMonthClick(month: string) {
  const eventKey = `click_kline_month:${month}`;

  if (shouldSkip(eventKey)) return;

  onSelectMonth(month);
  setDrawerOpen(true);
  track(...);
}
```

## 5. 行为记录本地失败队列

现在写行为失败只 `console.warn`，还不够。
至少补一个本地队列，防止反馈丢失。

```ts
// app/src/utils/behaviorQueue.ts

const STORAGE_KEY = 'life_state_behavior_queue';

export function enqueueBehavior(event: unknown) {
  const list = getBehaviorQueue();
  list.push(event);
  localStorage.setItem(STORAGE_KEY, JSON.stringify(list.slice(-100)));
}

export function getBehaviorQueue(): unknown[] {
  try {
    return JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]');
  } catch {
    return [];
  }
}

export function clearBehaviorQueue() {
  localStorage.removeItem(STORAGE_KEY);
}
```

在 `useBehaviorTracking` 里：

```ts
try {
  await consultApi.saveBehavior(recordId, body);
} catch (error) {
  enqueueBehavior({ recordId, body });
  console.warn('[behavior-track-queued]', eventType, error);
}
```

## 6. 命理副证冲突降级规则

命理副证不能跟主状态硬冲。
比如命理说“事业机会”，但状态图显示 `capacity低 + entropy高`，前台要自动降级表达。

```ts
// app/src/utils/metaphysicsConflict.ts

import type { PhasePoint } from '../types/lifekline';
import type { MetaphysicsHint } from '../types/tideClosedLoop';

export function downgradeMetaphysicsIfConflict(
  hint: MetaphysicsHint | undefined,
  phasePoint?: PhasePoint
): MetaphysicsHint | undefined {
  if (!hint?.show || !phasePoint) return hint;

  const highRiskPhase = phasePoint.quadrant === 'adjusting';

  if (!highRiskPhase) return hint;

  return {
    ...hint,
    confidence: Math.min(hint.confidence, 60),
    hintText: `命理副线仅作观察提示。当前现实状态处于调整区，主建议以降负荷和修复底盘为先。`,
  };
}
```

## 8. 后端 event_type 白名单

后端不能什么事件都收。
要补白名单，否则脏数据会进库。

```ts
// server/src/consult/behavior/behavior-event.types.ts

export const ALLOWED_BEHAVIOR_EVENTS = [
  'view_kline',
  'click_kline_month',
  'view_state_radar',
  'view_phase_chart',
  'click_tide_from_result',
  'claim_confirmed',
  'claim_partial',
  'claim_rejected',
  'action_executed',
  'action_partial',
  'action_skipped',
  'view_metaphysics_hint',
  'hide_metaphysics_hint',
] as const;

export type BehaviorEventType = typeof ALLOWED_BEHAVIOR_EVENTS[number];

export function isAllowedBehaviorEvent(value: string): value is BehaviorEventType {
  return ALLOWED_BEHAVIOR_EVENTS.includes(value as BehaviorEventType);
}
```

Service 里：

```ts
if (!isAllowedBehaviorEvent(dto.event_type)) {
  throw new BadRequestException(`Unsupported behavior event: ${dto.event_type}`);
}
```
